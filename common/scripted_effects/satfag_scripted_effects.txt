# Sets import and export tariff for the goods currently in scope to $level$,
# unless the country is already running any subvention on that good. Existing
# subsidy plans set up by the player are preserved.
# Called from country scope as: g:<good> = { set_tariffs_to_level = { level = <level> } }
set_tariffs_to_level = {
	if = {
		limit = {
			NOT = {
				root = { has_any_subventions_on = prev }
			}
		}
		root = {
			set_import_tariff_level = {
				goods = prev
				level = $level$
			}
			set_export_tariff_level = {
				goods = prev
				level = $level$
			}
		}
		custom_tooltip = set_tariffs_to_$level$_tt
	}
}
