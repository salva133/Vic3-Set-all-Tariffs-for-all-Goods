# Sets the import and export tariff for the market good currently in scope to
# $level$, unless the country (root) is already running a subvention on that
# good. Existing subsidy plans set up by the player are preserved.
# Expects a market_goods entry in scope (e.g. an element of every_market_goods),
# so the custom tooltip's [THIS.GetGoods...] resolves to the underlying good.
set_tariffs_to_level = {
	# Resolve the good behind this market entry once, so it can be referenced
	# from the country (root) scope below and stays stable across the block.
	goods = {
		save_scope_as = satfag_tariff_good
	}
	if = {
		limit = {
			NOT = {
				root = { has_any_subventions_on = scope:satfag_tariff_good }
			}
		}
		root = {
			set_import_tariff_level = {
				goods = scope:satfag_tariff_good
				level = $level$
			}
			set_export_tariff_level = {
				goods = scope:satfag_tariff_good
				level = $level$
			}
		}
		custom_tooltip = set_tariffs_to_$level$_tt
	}
}

# Applies the given tariff level to every tradeable good in the country's market,
# discovered dynamically through every_market_goods instead of a hardcoded list.
# This automatically covers goods added by other mods (any good that is present
# in the market), keeping the mod robust across goods overhauls. Goods that
# currently have a subvention are skipped by the inner helper.
# Called from country scope as: set_all_goods_to_level = { level = <level> }
set_all_goods_to_level = {
	market ?= {
		every_market_goods = {
			limit = {
				is_tradeable = yes
			}
			set_tariffs_to_level = { level = $level$ }
		}
	}
}
