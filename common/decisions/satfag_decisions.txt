decision_set_all_tariffs = {
	is_shown = {
		is_player = yes
	}
	possible = {
		# Only shown when the country actually controls its own market tariffs.
		# Customs union juniors cannot, but subjects outside a customs union can.
		is_junior_in_customs_union = no
		NOT = {
			has_variable = cooldown_bulk_set_tariffs_var
		}
	}
	when_taken = {
		trigger_event = {
			id = set_all_tariffs.2
			popup = yes
		}
	}
}
