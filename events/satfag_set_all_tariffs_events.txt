namespace = set_all_tariffs

set_all_tariffs.2 = {
	type = country_event
	placement = root

	title  = set_all_tariffs.2.tt
	desc   = set_all_tariffs.2.dd
	flavor = set_all_tariffs.2.ff

	event_image = {
		video = unspecific_signed_contract
	}

	trigger = {} # Gets triggered by decision_set_all_tariffs

	immediate = {}

	# Cooldown is applied no matter which option the player picks (including
	# cancel) so the decision cannot be spammed.
	after = {
		set_variable = {
			name = cooldown_bulk_set_tariffs_var
			months = 2
		}
	}

	option = {
		name = set_all_tariffs.2.a
		custom_tooltip = satfag_subsidies_preserved_tt
		set_all_goods_to_level = { level = no_tariffs_or_subventions }
	}
	option = {
		name = set_all_tariffs.2.b
		trigger = {
			NOT = { has_law = law_type:law_free_trade }
		}
		custom_tooltip = satfag_subsidies_preserved_tt
		set_all_goods_to_level = { level = low_tariffs }
	}
	option = {
		name = set_all_tariffs.2.c
		trigger = {
			NOT = { has_law = law_type:law_free_trade }
		}
		custom_tooltip = satfag_subsidies_preserved_tt
		set_all_goods_to_level = { level = high_tariffs }
	}
	option = {
		name = set_all_tariffs.2.d
		trigger = {
			NOT = { has_law = law_type:law_free_trade }
		}
		custom_tooltip = satfag_subsidies_preserved_tt
		set_all_goods_to_level = { level = max_tariffs }
	}
	option = {
		name = set_all_tariffs.2.e
		custom_tooltip = satfag_nothing_ever_happenes_tt
	}
}
